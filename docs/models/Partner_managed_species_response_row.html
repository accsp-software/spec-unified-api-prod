Partner_managed_species_response_row

  • attribute_code : Integer
    unique attribute code/id
  • attribute_name : String
    name of attribute
  • attribute_desc : String
    accompanying help text
  • species_itis : String
    ITIS code of species
  • type : String
    format of attribute.
    Enum:
    NUMBER*
    STRING
    DATE
    TIMESTAMP
  • list_type : String
    denotes whether list returned is static (S) or dynamic(D)
    Enum:
    S
    D
  • options : List
    JSON key-value pairs showing the return and display value of static list items respectively
  • order_seq : Integer
    display order of attribute relative to others
  • attribute_count : String
    expected number of values returned. default is Single(S). Multiple (M).
    Enum:
    S
    M
  • comm_flag : String
    indicates if attribute is required (R), optional (Y) or Not displayed (N) in commercial trip
    Enum:
    R
    Y
    N
  • for_hire_charter_flag : String
    indicates if attribute is required (R), optional (Y) or Not displayed (N) in for-hire charter trip
    Enum:
    R
    Y
    N
  • for_hire_hb_flag : String
    indicates if attribute is required (R), optional (Y) or Not displayed (N) in for-hire headboat trip
    Enum:
    R
    Y
    N
  • rec_flag : String
    indicates if attribute is required (R), optional (Y) or Not displayed (N) in recreational trip
    Enum:
    R
    Y
    N
  • limit_to_groups : partner_managed_species_response_limit_to_groups
